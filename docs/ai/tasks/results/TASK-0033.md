window.measureM2 = async function (targetDay) {
  const pause = () => new Promise(resolve => setTimeout(resolve, 550));
  const pick = async (selector, value) => {
    const el = document.querySelector(selector);
    el.value = value;
    el.dispatchEvent(new Event('change', { bubbles: true }));
    await pause();
  };
  const oldConfirm = window.confirm;
  window.confirm = () => true;
  try {
    await pick('[data-action="scenario"]', 'SCENARIO_M2_OPERATIONS');
  } finally { window.confirm = oldConfirm; }
  for (let day = 1; day < targetDay; day++) {
    if (targetDay === 15 && day === 14) {
      document.querySelector('[data-action="exchange"]').click();
      await pause();
    }
    document.querySelector('[data-action="end-day"]').click();
    await pause();

  if (targetDay === 15) {
    await pick('[data-action="offer-quantity"][data-key="MKT-D015-COS-BUY+MKT-D015-COS-SELL"]', '100');
  }
};
await measureM2(1);   // 상태 1 측정
await measureM2(8);   // 상태 2 측정

## 완료 조건 대조 (지시서의 조건마다 충족/미충족과 근거)

1. 범위 0~8: 충족. 규칙 2에서만 전용 칸을 표시하고 읽기 함수는 view로 호출한다. 수량 두 수락 길·거절·저장 두 경로·조건 문장 반례·never 분기·크기 판단을 시험했다.
2. 검사 실패 0: 전체 vitest·타입·빌드 통과. 자료 검사·공통 검수 14종도 최종 통과, 실패 0.
3. 기본 run 400회 바이트 동일: 충족, cmp 0와 동일 SHA-256.
4. 변형 시험 7개: 충족, 모두 의도한 assertion 실패. 원문 복사와 cmp로 되돌림 확인.
5. 결과 보고·측정 상태 세 개: 충족, 이 문서에 기록.

## 범위 밖 발견

- 자료 거래처 이름을 전부 뺀 설정 사본으로 fallback 통합 시험을 시도했을 때 엔진의 `src/engine/engine.ts:403`·484행 로그가 `find(...)!.nameKo`를 읽어 예외가 났다. 현재 자료에는 해당 이름이 모두 있다. 엔진은 수정하지 않고 표시 도우미의 없는 ID 반례로 검사했다.
- 기존 원화 장부의 ‘미지급 급여’·‘급여’ 중심 행은 규칙 2의 임차료·환전 장부 전체를 표현하지 않는다. TASK-0034의 결산·보고 범위다.
- 빈 계약 상태에서 첫 견적이 만료되면 기존 ‘처음부터로 다시 시작’ 문장이 남는다. 규칙 2는 뒤 묶음이 다시 공개되므로 문구 보강 후보다. 이번 지시서에 없는 기존 만료 안내는 수정하지 않았다.
